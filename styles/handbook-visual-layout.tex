% Compact visual models are integrated into the established strategy cards.
% Do not add independent floating duplicates. Strategy visuals may be enlarged within the bounded side panel for legibility.
\newcommand{\QRattach}[2]{\expandafter\def\csname QRvisual@#1\endcsname{#2}}
\newcommand{\QRplainlabel}{\textbf{\sffamily\color{teal}In plain terms.}}
\newcommand{\QRstrategyplain}[2]{%
  \ifcsname QRvisual@#1\endcsname
    \noindent
    \begin{minipage}[t]{.49\linewidth}
      \vspace{0pt}\QRplainlabel\ #2
    \end{minipage}\hfill
    \begin{minipage}[t]{.47\linewidth}
      \vspace{0pt}\centering
      \adjustbox{width=.98\linewidth,max totalheight=58pt}{\csname QRvisual@#1\endcsname}%
      \par\vspace{3pt}{\scriptsize\sffamily\color{muted}Visual model\par}
    \end{minipage}%
  \else
    \QRplainlabel\ #2%
  \fi}
% Inline panels are used only where there is no applicable strategy card.
% The minipage keeps artwork and its explanation together, in reading order.
\newcommand{\QRinline}[2]{%
  \par\addvspace{5pt}\noindent
  \begin{minipage}{\linewidth}
    \begin{minipage}[c]{.43\linewidth}\centering
      \adjustbox{max totalsize={\linewidth}{51pt}}{#1}%
    \end{minipage}\hfill
    \begin{minipage}[c]{.53\linewidth}
      \small\sffamily\color{muted}\textbf{Visual model.} #2
    \end{minipage}
  \end{minipage}\par\addvspace{5pt}}
\newcommand{\QRinlinepair}[3]{%
  \par\addvspace{5pt}\noindent
  \begin{minipage}{\linewidth}
    \centering
    \adjustbox{max totalsize={.45\linewidth}{51pt}}{#1}\hfill
    \adjustbox{max totalsize={.45\linewidth}{51pt}}{#2}%
    \par\smallskip{\small\sffamily\color{muted}\textbf{Visual model.} #3\par}
  \end{minipage}\par\addvspace{5pt}}
