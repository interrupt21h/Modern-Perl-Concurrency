\sect{1 / Choose the execution model}
\entry{Async / event loop}{Many network requests, timers; no blocking calls on the loop. State may change across an \code{await}.}
\entry{Ithreads}{Existing shared-state code or fixed worker pools. Load \code{threads} before \code{threads::shared}; interpreter cloning has memory cost. Discouraged for new designs.}
\entry{Fork / MCE}{CPU-heavy batches, isolation and bounded worker pools. Open DBI handles and sockets in the worker; do not reuse inherited connections.}
\sect{2 / Match the primitive to the job}
\begin{tabularx}{\linewidth}{@{}>{\bfseries\RaggedRight\arraybackslash}p{.30\linewidth} >{\RaggedRight\arraybackslash}X@{}}
Mutex & One owner; protect a compound update.\\
Semaphore & N permits; cap simultaneous work, not requests/sec.\\
Condition & Wait until a predicate is true; recheck after wake.\\
Barrier & All workers meet before the next phase.\\
Queue & Transfer ownership; bound backlog for backpressure.\\
RW lock & Many readers OR one writer; fairness matters.\\
Futex / atomic & Native implementation building blocks, not Perl scalar locks.\\
\end{tabularx}

\band{\texttt{\small use threads;\\use threads::shared;\\my \$n :shared = 0;\\\{ lock(\$n); ++\$n; \}}\\[-1pt]\textit{Excerpt: declare shared state before workers.}}
\alert{Every participant must lock the same stable shared variable. Container locks are not recursive locks on contents. Scope exit releases \code{lock()}.}

\sect{4 / Core Perl quick reference}
\entry{\code{threads::shared}}{\code{lock(\$gate)}; \code{shared\_clone} for nested data. No built-in timed \code{lock()}.}
\entry{\code{Thread::Semaphore}}{\code{new(3)}, \code{down}, \code{up}; balance permits on error. Use \code{down\_timed} for a deadline.}
\entry{\code{Thread::Queue}}{\code{enqueue}, \code{dequeue}, \code{end}; consumers drain then stop. Use a bounded queue when producers can outrun consumers.}
\entry{\code{cond\_wait}}{Hold the predicate's lock, wait inside a \code{while} loop; notifier changes state under the same lock and signals/broadcasts.}
\entry{\code{Future::Mutex}}{Serialize logical critical sections across suspension points; do not block the event loop.}
\sect{5 / Cross-process coordination}
\entry{\code{flock}}{Cooperating processes on one host: \code{LOCK\_EX} for writers, \code{LOCK\_SH} for readers, \code{LOCK\_NB} for nonblocking attempts. Use a dedicated stable lock file.}
\entry{\code{IPC::Semaphore}}{System V kernel permits shared by processes; manage creation, crash recovery and removal explicitly.}
\entry{Pipes / sockets}{Prefer sending work to one owner instead of coordinating writes to shared memory.}
\alert{A Perl thread lock is not a process lock. After \code{fork}, inherited file descriptions can share a \code{flock}; open separately per worker.}
\sect{6 / Design habits that prevent bugs}
\entry{Order}{Acquire multiple locks in one global order; release on every exit path.}
\entry{Duration}{Keep critical sections short. Avoid network I/O, sleeps and unbounded waits while holding a lock.}
\entry{Lifecycle}{Join workers; check failures. Stop admission, drain queues and shut down cleanly.}
\entry{Observability}{Measure wait time, hold time, timeout rate, queue depth; test forced interleavings and failures.}
\band{\textbf{Debugging shortcuts}\par\code{perl -V:useithreads}\par\code{ps -eLf}\enspace /\enspace \code{lslocks}\par\code{strace -f -e trace=futex,flock -p PID}\par\textit{Attach tracing only with authorization; it adds overhead.}}
\end{minipage}

\band{\textbf{Correctness vs. efficiency.} A duplicate cache refresh wastes work; a duplicate charge corrupts state. For the latter, enforce uniqueness, an atomic conditional update, or fencing at the destination. A lease alone is not a correctness guarantee.}

\sect{7 / MySQL + DBI / DBIx::Class}
\entry{Atomic update}{First choice for counters and guards. Check affected rows. A unique key prevents duplicate effects.}
\band{\texttt{UPDATE account\\SET cents = cents - 100\\WHERE id = 1 AND cents >= 100;}\\\textit{Require one affected row.}}
\entry{Pessimistic}{In a transaction, \code{SELECT ... FOR UPDATE} locks matching index records/ranges. Keep transactions short and indexed.}
\entry{Optimistic}{Version column in \code{WHERE}; increment it on update; retry or surface a conflict when zero rows change.}
\entry{Advisory}{\code{GET\_LOCK(name, timeout)} / \code{RELEASE\_LOCK(name)}; connection-owned, not transaction-owned. Check return values.}
\entry{Work queue}{\code{FOR UPDATE SKIP LOCKED} for claiming independent jobs; commit claims promptly. Use visibility deadlines, attempt IDs and idempotent effects.}
\entry{DBIx::Class}{Use \code{txn\_do} or \code{txn\_scope\_guard}; explicitly retry complete deadlocked transactions with bounded backoff.}
\alert{InnoDB may lock ranges, not just one row. Deadlocks are normal; rollback before retry. A network failure can leave commit outcome unknown.}
\sect{8 / Memcached: cache efficiency only}
\entry{Claim}{\code{add(key, owner, ttl)} succeeds only when absent.}
\entry{Release}{\code{gets} + owner check + \code{cas} short-lived tombstone, or let expire. Never unconditionally delete a successor's entry.}
\entry{Limitations}{Eviction, expiry, restart and rehash can invalidate ownership. No durable fencing; CAS IDs are not fence tokens.}
\sect{9 / Sessions and HTTP requests}
\entry{Prefer atomic state}{Session load/modify/save races across requests. Store counters/cart updates separately or serialize carefully.}
\entry{Idempotency}{Bind a key to request identity and payload; store result or pending state. A lock alone does not prevent repeat effects after a crash.}
\end{minipage}\hfill\begin{minipage}[t]{.485\linewidth}
\sect{10 / Redis / Valkey: owner-checked leases}
\band{\texttt{SET key token NX PX 10000}\\Acquire with a unique random owner token and TTL (milliseconds). Check success.}
\entry{Release / renew}{Compare the stored token and modify it atomically. Never use separate \code{GET} then \code{DEL}.}
\entry{Native commands}{Redis 8.4+: \code{DELEX ... IFEQ}; Valkey 9.0+: \code{DELIFEQ}. Conditional \code{SET ... IFEQ ... PX} renews on Redis 8.4+ / Valkey 8.1+.}
\entry{Perl transactions}{\code{WATCH key}, read owner, \code{MULTI}, queue update, \code{EXEC}. Retry confirmed conflicts only, using an exclusive connection.}
\entry{Lua invoked by Perl}{Server-side atomic check/update with \code{EVALSHA} or \code{EVAL}; not a separate application language. Cluster keys must share a slot.}

\entry{Fencing}{Assign monotonically increasing generations and reject stale generations at the protected resource. Preserve counter ordering across failures.}
\entry{Semaphore / queue}{Atomic permit bookkeeping with expiry; prefer Streams consumer groups for distributed jobs (acknowledge only after effect; reclaim abandoned work).}
\alert{TTL expiry does not stop a worker. An ambiguous renewal is not permission to continue. Replication and failover may lose lease state.}
\sect{11 / Decision shortcut}
\begin{tabularx}{\linewidth}{@{}>{\RaggedRight\arraybackslash}p{.39\linewidth}>{\RaggedRight\arraybackslash}X@{}}
\textbf{Need} & \textbf{Prefer}\\\hline
Shared durable data & DB constraint / transaction\\
Distributed duplicate work & Redis lease + bounded retry\\
Cache regeneration & Memcached add/CAS\\
Cross-host correctness & Authoritative coordinator + destination fencing\\
Durable job delivery & DB queue or durable broker; idempotency\\
\end{tabularx}
