\pagehead{1}{Local execution \& synchronization}{\textbf{Choose scope first:} async for I/O, fork/pools for CPU isolation, shared-memory primitives only for genuinely shared state. Solid arrows = progress; dashed arrows = wait, rejection or retry.}
\card{28}{114}{01}{Async event loop}{\Dasync}{Many I/O tasks interleave at suspension points; state can change across an await. Never block the loop.}{\api{IO::Async}, \api{Future::Mutex}}
\card{214}{114}{02}{Interpreter threads}{\Dthreads}{Perl clones interpreters; data sharing is explicit. Use a fixed pool for existing threaded code.}{\api{threads}, \api{threads::shared}}
\card{400}{114}{03}{Fork / worker pool}{\Dfork}{Separate address spaces. Good for CPU work and isolation. Open connections in each worker.}{\api{fork}, \api{MCE::Loop}}
\card{586}{114}{04}{Mutex}{\Dmutex}{One owner protects a compound update. All workers must lock the same shared variable.}{\api{lock(\$gate)}; scope releases}
\card{28}{261}{05}{Semaphore}{\Dsem}{N permits cap simultaneous work, not requests per second. Balance acquire and release.}{\api{Thread::Semaphore}: down / up}
\card{214}{261}{06}{Condition variable}{\Dcond}{Wait with predicate lock held; wait releases and reacquires it. Signal under lock; recheck in a loop.}{\api{cond\_wait}, \api{cond\_signal}}
\card{400}{261}{07}{Queue}{\Dqueue}{Transfer jobs to consumers instead of sharing mutable data. Bound backlog; drain on shutdown.}{\api{Thread::Queue}: enqueue / dequeue}
\card{586}{261}{08}{Barrier}{\Dbarrier}{Workers cannot enter the next phase until all arrive. A failed worker can stall everyone.}{Shared count + condition variable}
\card{28}{408}{09}{Reader / writer lock}{\Drw}{Many readers or one exclusive writer. Consider immutable snapshots; avoid writer starvation.}{Custom shared-state protocol}
\card{214}{408}{10}{File lock}{\Dflock}{Cooperating processes use a stable lock file. Open it independently after fork. Check success.}{\api{flock}: LOCK\_EX, LOCK\_SH, LOCK\_NB}
\card{400}{408}{11}{System V semaphore}{\Dsysv}{Kernel-managed permits across processes. Plan initialization, crash handling and removal.}{\api{IPC::Semaphore}; \api{semop}}
\card{586}{408}{12}{Atomics / futex}{\Dfutex}{Native synchronization building blocks. A Perl scalar is not a C atomic; do not emulate with polling.}{Native C / XS; Linux futex}
\pagefoot{\textbf{Before shipping:} name the invariant; use one global lock order; keep critical sections short; check worker errors; release on every path; test forced interleavings and measure wait/hold time.}

\pagehead{2}{Database \& distributed coordination}{\textbf{Correctness vs. efficiency:} duplicate refreshes waste work; duplicate charges corrupt state. Enforce durable invariants at the destination. A cache lease alone is not a correctness guarantee.}
\card{28}{114}{13}{Atomic SQL update}{\Datomic}{Put the invariant in the WHERE clause; check affected rows. Use unique keys to prevent duplicate effects.}{DBI / DBIx::Class conditional update}
\card{214}{114}{14}{Pessimistic row lock}{\Dpess}{Lock indexed records in a transaction. Range locks may occur; retry whole deadlocked transactions.}{\api{SELECT ... FOR UPDATE}}
\card{400}{114}{15}{Optimistic versioning}{\Dopt}{Read a version, then update only if it still matches. Increment version; zero rows means conflict.}{\api{WHERE version = ?}}
\card{586}{114}{16}{MySQL advisory lock}{\Dadvisory}{Named lock belongs to the connection, not the transaction. Check results and release explicitly.}{\api{GET\_LOCK} / \api{RELEASE\_LOCK}}
\card{28}{261}{17}{SQL work queue}{\Dskip}{Claim free jobs without waiting for held rows. Commit claim promptly; use attempt IDs and recovery.}{\api{FOR UPDATE SKIP LOCKED}}
\card{214}{261}{18}{Redis / Valkey lease}{\Dlease}{Use a unique owner token and TTL. Renew or release only by atomic owner check; never GET then DEL.}{\api{SET key token NX PX ttl}}
\card{400}{261}{19}{Optimistic Redis txn}{\Dwatch}{WATCH, read owner, MULTI, queue change, EXEC. Dedicated connection; retry confirmed conflicts only.}{Perl \api{Redis} client; WATCH / EXEC}
\card{586}{261}{20}{Redis server script}{\Dlua}{Perl invokes an atomic compare-and-change operation. Same hash slot for cluster keys.}{\api{EVALSHA} / \api{EVAL}; Lua inside Redis}
\card{28}{408}{21}{Fencing token}{\Dfence}{Issue durable, increasing generations. The destination rejects stale writes; expiry cannot stop a worker.}{Destination-enforced generation check}
\card{214}{408}{22}{Distributed permits / jobs}{\Ddistributed}{Atomically count permits with expiry. For Streams jobs, reclaim abandoned work and ACK after effect.}{Redis semaphore / Streams groups}
\card{400}{408}{23}{Memcached add / CAS}{\Dmem}{Best-effort cache stampede control. Owner-check then CAS a tombstone, or let expire. Never delete a successor.}{\api{add}, \api{gets}, \api{cas}; or let expire}
\card{586}{408}{24}{Sessions / idempotency}{\Didem}{Concurrent requests can lose session updates. Persist a request key, payload identity and outcome.}{Atomic state; unique key / saved result}
\pagefoot{\textbf{Before shipping:} define owner, deadline, retry budget, crash recovery and stale-writer policy. A network failure can leave commit outcome unknown; do not blindly repeat side effects.}
